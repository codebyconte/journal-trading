'use client'

import { useEffect, useState } from 'react'
import { BookOpen, Check, Download, Target, Calculator, ArrowRight } from 'lucide-react'
import { Card } from '@/components/ui/Card'
import { Button } from '@/components/catalyst/button'
import { PageShell } from '@/components/ui/PageShell'

const checks = ['J’ai identifié le contexte et les niveaux clés.', 'Mon setup et son invalidation sont définis.', 'Mon risque et mon exposition totale sont vérifiés.', 'Je suis disponible et calme pour exécuter mon plan.']
type Note = { scenario: string; invalidation: string; focus: string; lesson: string; next: string; checks: boolean[] }
type Store = Record<string, Note>
const blank = (): Note => ({ scenario: '', invalidation: '', focus: '', lesson: '', next: '', checks: [false, false, false, false] })
const STORAGE = 'tradinglog-workspace-v1'
function localDate() { const d = new Date(); return `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')}` }
function validStore(value: unknown): value is Store { return !!value && typeof value === 'object' && !Array.isArray(value) && Object.entries(value).every(([key,n]) => /^\d{4}-\d{2}-\d{2}$/.test(key) && n && ['scenario','invalidation','focus','lesson','next'].every(k => typeof n[k] === 'string') && Array.isArray(n.checks) && n.checks.length === 4 && n.checks.every((b: unknown) => typeof b === 'boolean')) }
export function ProgressWorkspace() {
  const [tab, setTab] = useState<'plan'|'review'|'risk'>('plan')
  const [date, setDate] = useState('')
  const [notes, setNotes] = useState<Store>({})
  const [ready, setReady] = useState(false)
  const [status, setStatus] = useState('Chargement…')
  const [capital, setCapital] = useState('')
  const [risk, setRisk] = useState('')
  const [entry, setEntry] = useState('')
  const [stop, setStop] = useState('')
  const [direction, setDirection] = useState('LONG')
  useEffect(() => { setDate(localDate()); try { const raw = localStorage.getItem(STORAGE); const parsed = raw ? JSON.parse(raw) : {}; if (!validStore(parsed)) throw new Error(); setNotes(parsed); setStatus('Stockage local prêt'); setReady(true) } catch { setStatus('Stockage indisponible ou données illisibles. Export possible pour les nouvelles notes.'); } }, [])
  const note = notes[date] ?? blank()
  function update(patch: Partial<Note>) { const next = {...notes, [date]: {...note, ...patch}}; setNotes(next); if (ready) { try { localStorage.setItem(STORAGE, JSON.stringify(next)); setStatus('Enregistré dans ce navigateur') } catch { setStatus('Échec de sauvegarde : exporte tes notes avant de quitter.') } } }
  function download() { const url = URL.createObjectURL(new Blob([JSON.stringify({version: 1, notes}, null, 2)], {type: 'application/json'})); const a = document.createElement('a'); a.href = url; a.download = `tradinglog-progression-${date}.json`; a.click(); setTimeout(() => URL.revokeObjectURL(url), 1000) }
  const c = Number(capital), r = Number(risk), e = Number(entry), sl = Number(stop)
  const valid = [c,r,e,sl].every(Number.isFinite) && c > 0 && r > 0 && r <= 100 && e > 0 && sl > 0 && (direction === 'LONG' ? sl < e : sl > e)
  const riskAmount = c*r/100, units = riskAmount/Math.abs(e-sl)
  const count = note.checks.filter(Boolean).length
  return <PageShell>
    <header className="flex flex-wrap items-end justify-between gap-4"><div><p className="eyebrow">APPRENDRE AVEC INTENTION</p><h1 className="mt-2 text-3xl font-semibold tracking-tight">Ma progression<span className="text-emerald-400">.</span></h1><p className="mt-2 text-sm text-zinc-400">Une préparation claire. Une leçon concrète. Une habitude à la fois.</p></div><Button outline onClick={download}><Download data-slot="icon"/> Exporter mes notes</Button></header>
    <div className="flex flex-wrap items-center justify-between gap-4 border-b border-white/10"><div role="tablist" aria-label="Espace de progression" className="flex flex-wrap gap-2">{([['plan','Préparation',Target],['review','Bilan hebdomadaire',BookOpen],['risk','Simulateur de risque',Calculator]] as const).map(([id,label,Icon]) => <button key={id} role="tab" id={`tab-${id}`} aria-selected={tab===id} aria-controls={`panel-${id}`} onClick={()=>setTab(id)} className={`flex items-center gap-2 border-b-2 px-3 py-4 text-sm ${tab===id?'border-emerald-400 text-emerald-300':'border-transparent text-zinc-400 hover:text-white'}`}><Icon size={16}/>{label}</button>)}</div>{tab !== 'risk' && <label className="flex items-center gap-2 pb-2 text-xs text-zinc-400">{tab === 'review' ? 'Date du bilan' : 'Séance'}<input aria-label="Date des notes" type="date" value={date} onChange={e=>{if(e.target.value) setDate(e.target.value)}} className="work-input max-w-44"/></label>}</div>
    <div role="tabpanel" id={`panel-${tab}`} aria-labelledby={`tab-${tab}`}>
    {tab==='plan' && <div className="grid gap-5 lg:grid-cols-[1.5fr_1fr]"><Card><p className="eyebrow">01 / ÉCRIRE AVANT D’AGIR</p><h2 className="mb-6 mt-2 text-xl">Mon plan de séance</h2>{([['scenario','Quel scénario est-ce que j’attends ?','Contexte, actif, unité de temps, zone et déclencheur…'],['invalidation','Qu’est-ce qui invalide mon scénario ?','Niveau, changement de contexte ou condition d’abstention…'],['focus','Mon engagement de processus','Exemple : attendre la clôture de ma bougie de référence.']] as const).map(([key,label,placeholder])=><label key={key} className="mb-5 block text-sm text-zinc-300">{label}<textarea className="work-input mt-2 min-h-24 resize-y" placeholder={placeholder} value={note[key]} onChange={e=>update({[key]:e.target.value})}/></label>)}</Card><div className="space-y-5"><Card><p className="eyebrow">02 / VÉRIFIER SON ÉTAT</p><h2 className="mt-2 text-xl">Checklist de préparation</h2><p className="mt-2 text-sm text-zinc-400">{count} / 4 points vérifiés</p><div className="my-5 h-1.5 rounded-full bg-white/10"><div className="h-full rounded-full bg-emerald-400 transition-all" style={{width:`${count*25}%`}}/></div>{checks.map((label,i)=><label key={label} className="flex cursor-pointer items-start gap-3 border-t border-white/5 py-4 text-sm leading-6 text-zinc-300"><input type="checkbox" checked={note.checks[i]} onChange={e=>update({checks:note.checks.map((v,j)=>i===j?e.target.checked:v)})} className="mt-1 size-4 accent-emerald-400"/>{label}</label>)}</Card><div className="rounded-xl border border-emerald-400/15 bg-emerald-400/5 p-5"><p className="text-sm text-emerald-200">Ne pas trader est aussi une décision.</p><p className="mt-2 text-xs leading-5 text-zinc-400">Cette checklist complète les contrôles du formulaire de trade. Elle ne remplace pas ton protocole.</p></div></div></div>}
    {tab==='review' && <div className="grid gap-5 lg:grid-cols-[1.5fr_1fr]"><Card><p className="eyebrow">PRENDRE DU RECUL</p><h2 className="mb-6 mt-2 text-xl">Transformer une observation en action</h2><label className="block text-sm">Ma leçon de la semaine<textarea className="work-input mt-3 min-h-40" value={note.lesson} onChange={e=>update({lesson:e.target.value})} placeholder="Quel trade ai-je bien exécuté, même perdant ? Quelle erreur se répète ? Appuie-toi sur des exemples précis."/></label><label className="mt-6 block text-sm">Une action mesurable pour la semaine prochaine<textarea className="work-input mt-3 min-h-28" value={note.next} onChange={e=>update({next:e.target.value})} placeholder="Exemple : documenter mon invalidation avant chaque entrée."/></label></Card><Card><h2 className="font-semibold">Mes bilans enregistrés</h2><p className="mt-2 text-xs text-zinc-400">Sélectionne une date pour reprendre un bilan.</p><div className="mt-5 space-y-2">{Object.entries(notes).filter(([,n])=>n.lesson || n.next).sort(([a],[b])=>b.localeCompare(a)).map(([day,n])=><button key={day} onClick={()=>setDate(day)} className={`w-full rounded-lg border p-3 text-left ${date===day?'border-emerald-400/40 bg-emerald-400/5':'border-white/10 hover:bg-white/5'}`}><span className="text-xs text-emerald-300">{day}</span><p className="mt-1 truncate text-sm text-zinc-300">{n.lesson || n.next}</p></button>)}{!Object.values(notes).some(n=>n.lesson||n.next) && <p className="py-8 text-sm text-zinc-500">Ton premier bilan apparaîtra ici.</p>}</div></Card></div>}
    {tab==='risk' && <div className="grid gap-5 lg:grid-cols-2"><Card><p className="eyebrow">SIMULER AVANT L’ENTRÉE</p><h2 className="mb-6 mt-2 text-xl">Dimensionner une position</h2><label className="mb-4 block text-sm">Direction<select className="work-input mt-2" value={direction} onChange={e=>setDirection(e.target.value)}><option>LONG</option><option>SHORT</option></select></label><div className="grid gap-4 sm:grid-cols-2">{[['Capital de référence',capital,setCapital],['Risque choisi (%)',risk,setRisk],['Prix d’entrée',entry,setEntry],['Prix du stop',stop,setStop]].map(([label,value,setter])=><label key={String(label)} className="block text-sm">{String(label)}<input type="number" min="0" step="any" className="work-input mt-2" value={String(value)} onChange={e=>(setter as (v:string)=>void)(e.target.value)} placeholder="0"/></label>)}</div></Card><Card><p className="eyebrow">RÉSULTAT DE LA SIMULATION</p><div aria-live="polite">{valid ? <div className="mt-6 space-y-6">{[['Perte théorique au stop',riskAmount.toLocaleString('fr-FR',{maximumFractionDigits:2})],['Quantité théorique',units.toLocaleString('fr-FR',{maximumFractionDigits:8})],['Valeur notionnelle',(units*e).toLocaleString('fr-FR',{maximumFractionDigits:2})]].map(([label,value])=><div key={label}><p className="text-sm text-zinc-400">{label}</p><p className="mt-2 font-mono text-3xl text-emerald-300">{value}</p></div>)}</div> : <p className="my-10 text-sm leading-6 text-zinc-400">Renseigne quatre valeurs positives. Le stop doit être sous l’entrée pour un long, au-dessus pour un short. Le risque choisi doit être compris entre 0 et 100 % exclus pour 0.</p>}</div><p className="mt-6 border-t border-white/10 pt-4 text-xs leading-5 text-zinc-400">Calcul pour un instrument linéaire : quantité = risque monétaire ÷ distance au stop. Montants dans la devise du capital et des prix, qui doivent être identiques. Hors frais, glissement, multiplicateurs de contrats et liquidation. Aucune opération n’est enregistrée.</p></Card></div>}
    </div>
    <footer className="flex flex-wrap items-center justify-between gap-3 text-xs text-zinc-500"><p role="status" className="flex items-center gap-2"><Check size={14}/>{status}</p><p>Notes privées à ce navigateur · pense à les exporter.</p></footer>
  </PageShell>
}
